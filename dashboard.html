-- Function untuk memulai izin dengan atomic check CREATE OR REPLACE FUNCTION start_izin( p_nama TEXT, p_role TEXT, p_tanggal TEXT, p_mulai TIMESTAMPTZ ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_limit INTEGER; v_current_count INTEGER; v_result JSONB; BEGIN -- Ambil limit dari settings SELECT CASE WHEN p_role = 'cs' THEN (SELECT COALESCE((value::jsonb->>'cs')::int, 2) FROM settings WHERE key = 'role_limits') WHEN p_role = 'kapten' THEN (SELECT COALESCE((value::jsonb->>'kapten')::int, 1) FROM settings WHERE key = 'role_limits') WHEN p_role = 'kasir' THEN (SELECT COALESCE((value::jsonb->>'kasir')::int, 2) FROM settings WHERE key = 'role_limits') ELSE 99 END INTO v_limit; -- Hitung user yang sedang izin untuk role yang sama SELECT COUNT(*) INTO v_current_count FROM izin_anggota WHERE role = p_role AND status = 'izin'; -- Cek apakah masih ada kuota IF v_current_count >= v_limit AND v_limit < 99 THEN v_result := jsonb_build_object( 'success', false, 'message', 'Kuota penuh untuk role ' || p_role || '. Saat ini ' || v_current_count || ' dari ' || v_limit || ' sedang izin.', 'current_count', v_current_count, 'limit', v_limit ); RETURN v_result; END IF; -- Cek apakah user sudah izin IF EXISTS (SELECT 1 FROM izin_anggota WHERE nama = p_nama AND status = 'izin') THEN v_result := jsonb_build_object( 'success', false, 'message', 'User ' || p_nama || ' sedang dalam izin!', 'current_count', v_current_count, 'limit', v_limit ); RETURN v_result; END IF; -- Insert atau update data izin INSERT INTO izin_anggota (nama, status, mulai, tanggal, role) VALUES (p_nama, 'izin', p_mulai, p_tanggal, p_role) ON CONFLICT (nama) DO UPDATE SET status = 'izin', mulai = p_mulai, tanggal = p_tanggal, role = p_role; -- Hitung ulang setelah insert SELECT COUNT(*) INTO v_current_count FROM izin_anggota WHERE role = p_role AND status = 'izin'; v_result := jsonb_build_object( 'success', true, 'message', 'Berhasil memulai izin untuk ' || p_nama, 'current_count', v_current_count, 'limit', v_limit ); RETURN v_result; END; $$;
